【1】GameInitializer - 游戏启动管理器
====================================

位置：Assets/Scripts/Game/GameInitializer.cs

```csharp
using UnityEngine;
using UnityEngine.SceneManagement;
using System.Collections;

/// <summary>
/// 游戏初始化器 - 负责游戏启动和初始化
/// </summary>
public class GameInitializer : MonoBehaviour
{
    [SerializeField] private GameManager _gameManager;
    [SerializeField] private UIManager _uiManager;

    private void Awake()
    {
        // 初始化各大系统
        InitializeSystems();
    }

    private void Start()
    {
        // 启动游戏主流程
        StartCoroutine(InitializeGame());
    }

    private void InitializeSystems()
    {
        Debug.Log("[GameInitializer] 初始化系统");

        if (_gameManager == null)
            _gameManager = FindObjectOfType<GameManager>();

        if (_uiManager == null)
            _uiManager = FindObjectOfType<UIManager>();

        // 初始化DataManager
        var dataManager = _gameManager?.DataManager;
        if (dataManager != null)
        {
            dataManager.Initialize();
            dataManager.LoadPlayerData();
        }

        // 初始化配置加载器
        TribeSystem.TribeConfigLoader.Instance.LoadAllConfigs();
    }

    private IEnumerator InitializeGame()
    {
        yield return null; // 等一帧让所有初始化完成

        var dataManager = _gameManager?.DataManager;
        if (dataManager == null)
        {
            Debug.LogError("[GameInitializer] DataManager not found");
            yield break;
        }

        // 检查是否是新游戏
        bool isNewGame = dataManager.GetCurrentRound() == 1 &&
                        (dataManager.GetTribes() == null || dataManager.GetTribes().Count == 0);

        if (isNewGame)
        {
            Debug.Log("[GameInitializer] 新游戏 - 显示初始族群选择");
            ShowInitialTribeSelection();
        }
        else
        {
            Debug.Log("[GameInitializer] 继续游戏 - 进入TribeBuildPanel");
            EnterGameMain();
        }
    }

    /// <summary>
    /// 显示初始族群选择界面
    /// </summary>
    private void ShowInitialTribeSelection()
    {
        if (_uiManager == null) return;

        var panel = _uiManager.ShowPanel<InitialTribeSelectionPanel>(
            "ui/initialtribeselection",
            UIManager.UILayer.Normal
        );

        if (panel != null)
        {
            panel.OnSelectionComplete += EnterGameMain;
        }
    }

    /// <summary>
    /// 进入游戏主界面
    /// </summary>
    private void EnterGameMain()
    {
        Debug.Log("[GameInitializer] 进入游戏主界面");

        if (_uiManager == null) return;

        var panel = _uiManager.ShowPanel<TribeSystem.UI.TribeBuildPanel>(
            "ui/tribebuild/tribebuildpanel",
            UIManager.UILayer.Normal
        );

        if (panel != null)
        {
            Debug.Log("[GameInitializer] TribeBuildPanel已显示");
        }
    }
}
```

关键点：
- Initialize() 在Awake中初始化所有系统
- LoadPlayerData() 加载玩家数据
- 检查是否是新游戏
- 根据情况显示初始选择或主游戏界面

==================================================

位置：Assets/Scripts/UI/InitialTribeSelectionPanel.cs

```csharp
using UnityEngine;
using UnityEngine.UI;
using System;
using System.Collections.Generic;
using TribeSystem;
using TribeSystem.UI;

/// <summary>
/// 初始族群选择面板 - 六选二
/// 玩家选择2个族群作为初始战力
/// </summary>
public class InitialTribeSelectionPanel : UIPanel
{
    private const string PanelName = "初始族群选择";
    private const string HintText = "选择2个族群作为初始战力（已选：0/2）";

    [Header("UI 组件")]
    [SerializeField] private Text _titleText;
    [SerializeField] private Text _hintText;
    [SerializeField] private RectTransform _tribesContainer;
    [SerializeField] private Button _confirmButton;

    // 事件
    public event Action OnSelectionComplete;

    // 状态
    private List<TribeType> _selectedTribes = new List<TribeType>();
    private Dictionary<TribeType, bool> _tribeSelectionState = new Dictionary<TribeType, bool>();
    private Font _cachedFont;

    public override void Initialize()
    {
        base.Initialize();
        Debug.Log("[InitialTribeSelectionPanel] Initialized");

        EnsureUIComponents();
        InitializeTribeButtons();
    }

    private void EnsureUIComponents()
    {
        _cachedFont = Resources.GetBuiltinResource<Font>("LegacyRuntime.ttf");

        if (_titleText == null)
            _titleText = transform.Find("Title")?.GetComponent<Text>();
        if (_hintText == null)
            _hintText = transform.Find("Hint")?.GetComponent<Text>();
        if (_tribesContainer == null)
            _tribesContainer = transform.Find("TribesContainer") as RectTransform;
        if (_confirmButton == null)
            _confirmButton = transform.Find("ConfirmButton")?.GetComponent<Button>();

        // 设置UI文本
        if (_titleText != null)
            _titleText.text = PanelName;
        if (_confirmButton != null)
        {
            _confirmButton.onClick.AddListener(OnConfirmClicked);
            _confirmButton.interactable = false;
        }

        UpdateHintText();
    }

    private void InitializeTribeButtons()
    {
        if (_tribesContainer == null) return;

        // 清空容器
        foreach (Transform child in _tribesContainer)
        {
            Destroy(child.gameObject);
        }

        // 为每个族群创建按钮
        foreach (TribeType type in System.Enum.GetValues(typeof(TribeType)))
        {
            CreateTribeButton(type);
            _tribeSelectionState[type] = false;
        }
    }

    private void CreateTribeButton(TribeType tribeType)
    {
        GameObject btnGo = new GameObject($"Tribe_{tribeType}", typeof(RectTransform), typeof(Button), typeof(Image));
        btnGo.transform.SetParent(_tribesContainer, false);

        RectTransform btnRect = btnGo.GetComponent<RectTransform>();
        btnRect.sizeDelta = new Vector2(200f, 250f);

        Image btnImg = btnGo.GetComponent<Image>();
        btnImg.color = GetTribeColor(tribeType);

        Button btn = btnGo.GetComponent<Button>();
        btn.onClick.AddListener(() => OnTribeButtonClicked(tribeType));

        // 创建按钮内容
        CreateTribeButtonContent(btnRect, tribeType);
    }

    private void CreateTribeButtonContent(RectTransform btnRect, TribeType tribeType)
    {
        // 族群名称
        GameObject nameGo = new GameObject("Name", typeof(RectTransform), typeof(Text));
        nameGo.transform.SetParent(btnRect, false);
        RectTransform nameRect = nameGo.GetComponent<RectTransform>();
        nameRect.anchorMin = new Vector2(0.1f, 0.75f);
        nameRect.anchorMax = new Vector2(0.9f, 0.90f);
        nameRect.offsetMin = Vector2.zero;
        nameRect.offsetMax = Vector2.zero;
        Text nameText = nameGo.GetComponent<Text>();
        nameText.font = _cachedFont;
        nameText.fontSize = 18;
        nameText.alignment = TextAnchor.MiddleCenter;
        nameText.color = Color.white;
        nameText.text = GetTribeTypeName(tribeType);

        // 族群描述
        GameObject descGo = new GameObject("Description", typeof(RectTransform), typeof(Text));
        descGo.transform.SetParent(btnRect, false);
        RectTransform descRect = descGo.GetComponent<RectTransform>();
        descRect.anchorMin = new Vector2(0.1f, 0.30f);
        descRect.anchorMax = new Vector2(0.9f, 0.70f);
        descRect.offsetMin = Vector2.zero;
        descRect.offsetMax = Vector2.zero;
        Text descText = descGo.GetComponent<Text>();
        descText.font = _cachedFont;
        descText.fontSize = 12;
        descText.alignment = TextAnchor.MiddleCenter;
        descText.color = new Color(0.9f, 0.9f, 0.9f, 1f);
        descText.text = GetTribeDescription(tribeType);

        // 初始小猫数
        var config = TribeConfigLoader.Instance.GetTribeConfig(tribeType);
        GameObject countGo = new GameObject("Count", typeof(RectTransform), typeof(Text));
        countGo.transform.SetParent(btnRect, false);
        RectTransform countRect = countGo.GetComponent<RectTransform>();
        countRect.anchorMin = new Vector2(0.1f, 0.10f);
        countRect.anchorMax = new Vector2(0.9f, 0.25f);
        countRect.offsetMin = Vector2.zero;
        countRect.offsetMax = Vector2.zero;
        Text countText = countGo.GetComponent<Text>();
        countText.font = _cachedFont;
        countText.fontSize = 14;
        countText.alignment = TextAnchor.MiddleCenter;
        countText.color = new Color(1f, 0.9f, 0.3f, 1f);
        countText.text = config != null ? $"初始小猫: {config.initialCatCount}只" : "初始小猫: ?只";
    }

    private void OnTribeButtonClicked(TribeType tribeType)
    {
        bool isCurrentlySelected = _selectedTribes.Contains(tribeType);

        if (isCurrentlySelected)
        {
            // 取消选择
            _selectedTribes.Remove(tribeType);
            _tribeSelectionState[tribeType] = false;
        }
        else
        {
            // 选择（最多2个）
            if (_selectedTribes.Count < 2)
            {
                _selectedTribes.Add(tribeType);
                _tribeSelectionState[tribeType] = true;
            }
            else
            {
                Debug.Log("[InitialTribeSelectionPanel] 最多选择2个族群");
                return;
            }
        }

        // 更新UI
        UpdateTribeButtonStates();
        UpdateHintText();
        UpdateConfirmButton();
    }

    private void UpdateTribeButtonStates()
    {
        for (int i = 0; i < _tribesContainer.childCount; i++)
        {
            Transform child = _tribesContainer.GetChild(i);
            Image childImg = child.GetComponent<Image>();

            // 从按钮名称解析族群类型
            string btnName = child.gameObject.name;
            if (btnName.StartsWith("Tribe_"))
            {
                string typeStr = btnName.Substring(6);
                if (System.Enum.TryParse<TribeType>(typeStr, out TribeType type))
                {
                    if (_selectedTribes.Contains(type))
                    {
                        // 高亮显示
                        childImg.color = new Color(1f, 0.9f, 0.3f, 1f);
                    }
                    else
                    {
                        // 恢复原色
                        childImg.color = GetTribeColor(type);
                    }
                }
            }
        }
    }

    private void UpdateHintText()
    {
        if (_hintText != null)
        {
            _hintText.text = $"选择2个族群作为初始战力（已选：{_selectedTribes.Count}/2）";
        }
    }

    private void UpdateConfirmButton()
    {
        if (_confirmButton != null)
        {
            _confirmButton.interactable = _selectedTribes.Count == 2;
        }
    }

    private void OnConfirmClicked()
    {
        if (_selectedTribes.Count != 2)
        {
            Debug.LogWarning("[InitialTribeSelectionPanel] 请选择2个族群");
            return;
        }

        Debug.Log($"[InitialTribeSelectionPanel] 选择族群: {_selectedTribes[0]}, {_selectedTribes[1]}");

        // 创建初始族群并保存
        CreateAndSaveInitialTribes();

        // 触发完成事件
        OnSelectionComplete?.Invoke();

        // 关闭面板
        Hide();
    }

    private void CreateAndSaveInitialTribes()
    {
        var dataManager = GameManager.Instance?.DataManager;
        if (dataManager == null) return;

        foreach (var tribeType in _selectedTribes)
        {
            var config = TribeConfigLoader.Instance.GetTribeConfig(tribeType);
            if (config == null) continue;

            var tribe = new TribeRecord
            {
                tribeId = dataManager.GetTribes()?.Count ?? 0,
                tribeType = tribeType,
                leader = new LeaderData
                {
                    leaderId = Random.Range(1000, 9999),
                    name = $"{config.tribeName}族长",
                    baseAttack = config.leaderBaseStats.attack,
                    baseDefense = config.leaderBaseStats.defense,
                    baseHp = config.leaderBaseStats.hp,
                    baseSpeed = config.leaderBaseStats.speed,
                    command = config.leaderBaseStats.command,
                    skillIds = new List<int>(),
                    permanentBuffs = new PermanentBuffs(),
                    restTurns = 0
                },
                cats = new List<CatData>(),
                isActive = true
            };

            // 添加初始小猫（白色品质）
            for (int i = 0; i < config.initialCatCount; i++)
            {
                tribe.cats.Add(CatData.CreateWithQuality(CatQuality.White));
            }

            dataManager.AddTribe(tribe);
        }

        dataManager.SavePlayerData();
    }

    #region Helper Methods

    private Color GetTribeColor(TribeType type)
    {
        return type switch
        {
            TribeType.Maine => new Color(0.3f, 0.5f, 0.7f, 1f),
            TribeType.Tabby => new Color(0.6f, 0.4f, 0.3f, 1f),
            TribeType.Orange => new Color(0.7f, 0.5f, 0.2f, 1f),
            TribeType.Cow => new Color(0.4f, 0.4f, 0.5f, 1f),
            TribeType.Siamese => new Color(0.5f, 0.4f, 0.6f, 1f),
            TribeType.Ragdoll => new Color(0.7f, 0.5f, 0.6f, 1f),
            _ => new Color(0.5f, 0.5f, 0.5f, 1f)
        };
    }

    private string GetTribeTypeName(TribeType type)
    {
        return type switch
        {
            TribeType.Maine => "缅因猫族",
            TribeType.Tabby => "狸花猫族",
            TribeType.Orange => "大橘猫族",
            TribeType.Cow => "奶牛猫族",
            TribeType.Siamese => "暹罗猫族",
            TribeType.Ragdoll => "布偶猫族",
            _ => type.ToString()
        };
    }

    private string GetTribeDescription(TribeType type)
    {
        return type switch
        {
            TribeType.Maine => "均衡型\n攻防兼备",
            TribeType.Tabby => "攻击型\n高攻低防",
            TribeType.Orange => "坦克型\n高血厚防",
            TribeType.Cow => "防御型\n防御最高",
            TribeType.Siamese => "敏捷型\n速度最快",
            TribeType.Ragdoll => "特殊型\n独特优势",
            _ => "???"
        };
    }

    #endregion
}
```

位置：Assets/Scripts/Game/GameFlowController.cs

```csharp
using UnityEngine;
using System;
using TribeSystem;
using TribeSystem.UI;

/// <summary>
/// 游戏流程控制器 - 管理整个游戏的流程
/// 包括回合循环、系统触发、界面流转等
/// </summary>
public class GameFlowController : MonoBehaviour
{
    public static GameFlowController Instance { get; private set; }

    [SerializeField] private GameManager _gameManager;
    [SerializeField] private UIManager _uiManager;
    [SerializeField] private TribeBuildPanel _tribeBuildPanel;

    private DataManager _dataManager;
    private RoundManager _roundManager;
    private bool _isGameRunning = false;

    // 事件
    public event Action OnGameStarted;
    public event Action OnRoundStarted;
    public event Action OnRoundEnded;
    public event Action OnGameEnded;

    private void Awake()
    {
        if (Instance != null && Instance != this)
        {
            Destroy(gameObject);
            return;
        }
        Instance = this;
        DontDestroyOnLoad(gameObject);
    }

    private void Start()
    {
        Initialize();
    }

    private void Initialize()
    {
        Debug.Log("[GameFlowController] 初始化");

        _gameManager = GameManager.Instance ?? FindObjectOfType<GameManager>();
        _uiManager = _gameManager?.UIManager;
        _dataManager = _gameManager?.DataManager;

        if (_dataManager == null)
        {
            Debug.LogError("[GameFlowController] DataManager not found");
            return;
        }

        _roundManager = new RoundManager();
        int savedRound = _dataManager.GetCurrentRound();
        _roundManager.SetRound(savedRound);
    }

    /// <summary>
    /// 开始游戏
    /// </summary>
    public void StartGame()
    {
        if (_isGameRunning) return;

        Debug.Log("[GameFlowController] 游戏开始");
        _isGameRunning = true;
        OnGameStarted?.Invoke();

        StartRound();
    }

    /// <summary>
    /// 开始新回合
    /// </summary>
    public void StartRound()
    {
        if (!_isGameRunning) return;

        Debug.Log($"[GameFlowController] 开始第{_roundManager.CurrentRound}回合");
        _roundManager.StartRound();
        OnRoundStarted?.Invoke();

        // 进入回合准备（由TribeBuildPanel处理）
        if (_tribeBuildPanel != null)
        {
            _tribeBuildPanel.StartRoundPreparation();
        }
    }

    /// <summary>
    /// 结束当前回合
    /// </summary>
    public void EndRound()
    {
        if (!_isGameRunning) return;

        _roundManager.EndRound();
        OnRoundEnded?.Invoke();

        if (_roundManager.IsGameOver)
        {
            EndGame();
        }
        else
        {
            StartRound();
        }
    }

    /// <summary>
    /// 游戏结束
    /// </summary>
    public void EndGame()
    {
        Debug.Log("[GameFlowController] 游戏结束");
        _isGameRunning = false;
        OnGameEnded?.Invoke();

        // 显示结算界面
        ShowGameOverScreen();
    }

    private void ShowGameOverScreen()
    {
        if (_uiManager == null) return;

        // TODO: 显示结算界面
        Debug.Log("[GameFlowController] 显示游戏结束界面");
    }

    public int GetCurrentRound() => _roundManager?.CurrentRound ?? 1;
    public bool IsGameRunning => _isGameRunning;
    public RoundManager RoundManager => _roundManager;
}
```
