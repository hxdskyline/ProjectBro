   ```csharp
   private void EnsureUIComponents()
   {
       // 如果预制体上已绑定，则不需要创建
       if (_titleText != null && _optionsContainer != null) return;

       // 这里可以通过查找子组件或报错处理
       if (_titleText == null) Debug.LogError("TitleText not assigned");
   }

   private void EnsureRuntimeUI(RectTransform parent, Font font)
   {
       // 如果已创建过，则跳过
       if (_isRuntimeCreated) return;

       // 创建Title
       GameObject titleGo = new GameObject("Title", typeof(RectTransform), typeof(Text));
       titleGo.transform.SetParent(parent, false);
       _titleText = titleGo.GetComponent<Text>();
       _titleText.text = PanelName;
       _titleText.font = font;

       // 创建Hint
       GameObject hintGo = new GameObject("Hint", typeof(RectTransform), typeof(Text));
       hintGo.transform.SetParent(parent, false);
       _hintText = hintGo.GetComponent<Text>();
       _hintText.text = HintForce;
       _hintText.font = font;

       // 创建选项容器
       GameObject containerGo = new GameObject("OptionsContainer", typeof(RectTransform), typeof(LayoutGroup));
       containerGo.transform.SetParent(parent, false);
       _optionsContainer = containerGo.GetComponent<RectTransform>();

       // 创建确认按钮
       GameObject confirmGo = new GameObject("ConfirmButton", typeof(RectTransform), typeof(Button), typeof(Image));
       confirmGo.transform.SetParent(parent, false);
       _confirmButton = confirmGo.GetComponent<Button>();
       _confirmButton.onClick.AddListener(OnConfirmClicked);

       _isRuntimeCreated = true;
   }

   private void GenerateOptionCards(List<RecruitmentOption> options)
   {
       // 遍历选项，为每个选项创建卡片
       for (int i = 0; i < options.Count; i++)
       {
           CreateOptionCard(options[i], i);
       }
   }

   private void CreateOptionCard(RecruitmentOption option, int index)
   {
       // 创建卡片背景
       GameObject cardGo = new GameObject($"OptionCard_{index}",
           typeof(RectTransform),
           typeof(Image),
           typeof(Button),
           typeof(Toggle));

       cardGo.transform.SetParent(_optionsContainer, false);

       Button cardButton = cardGo.GetComponent<Button>();
       Toggle cardToggle = cardGo.GetComponent<Toggle>();
       Image cardImage = cardGo.GetComponent<Image>();

       // 设置卡片外观
       cardImage.color = new Color(0.3f, 0.3f, 0.3f, 1f);
       cardToggle.group = _optionsContainer.GetComponent<ToggleGroup>();

       // 绑定点击事件
       cardButton.onClick.AddListener(() => OnOptionCardClicked(option, cardToggle));

       // 创建卡片内容
       Font font = _cachedFont != null ? _cachedFont : Resources.GetBuiltinResource<Font>("LegacyRuntime.ttf");

       // 选项名称
       CreateTextChild(cardGo, "Name", GetOptionTypeName(option.optionType), font, 14);

       // 成本显示
       CreateTextChild(cardGo, "Cost", $"消耗: {option.cost}猫粮", font, 12);

       // 描述
       CreateTextChild(cardGo, "Description", option.description, font, 10);
   }

   private Text CreateTextChild(GameObject parent, string name, string content, Font font, int fontSize)
   {
       GameObject textGo = new GameObject(name, typeof(RectTransform), typeof(Text));
       textGo.transform.SetParent(parent.transform, false);

       Text text = textGo.GetComponent<Text>();
       text.text = content;
       text.font = font;
       text.fontSize = fontSize;
       text.alignment = TextAnchor.MiddleCenter;

       return text;
   }

   private void OnOptionCardClicked(RecruitmentOption option, Toggle toggle)
   {
       _selectedOption = option;
       toggle.SetIsOnWithoutNotify(true);
       _confirmButton.interactable = true;
   }

   private void OnConfirmClicked()
   {
       if (_selectedOption != null)
       {
           _onOptionSelected?.Invoke(_selectedOption);
           Hide();
       }
   }

   private string GetOptionTypeName(RecruitmentOptionType type)
   {
       return type switch
       {
           RecruitmentOptionType.NewTribe => "新增族群",
           RecruitmentOptionType.AddCats => "增加小猫",
           RecruitmentOptionType.QualityEvolution => "品质进化",
           RecruitmentOptionType.LeaderBoost => "族长强化",
           _ => type.ToString()
       };
   }

   public void Hide()
   {
       gameObject.SetActive(false);
   }

   public void Show()
   {
       gameObject.SetActive(true);
   }

   private void UpdateHintText()
   {
       if (_hintText != null)
       {
           _hintText.text = HintForce;
       }
   }

   private void ClearOptionCards()
   {
       // 销毁容器内的所有卡片
       for (int i = _optionsContainer.childCount - 1; i >= 0; i--)
       {
           Destroy(_optionsContainer.GetChild(i).gameObject);
       }
   }
   ```
